## 烤烘

![image.png](https://picblog.oss-cn-hangzhou.aliyuncs.com/obsidian/20241209102908.png)

```sql
SELECT
	n.instance_id
FROM
	named_object n
WHERE
	n.instance_rrn IN (
	SELECT
		r.from_rrn
	FROM
		relation r
	WHERE
		r.link_type = 'ENTITY_TO_ENTITY_GROUP'
		AND r.to_rrn = (
		SELECT
			n.instance_rrn
		FROM
			named_object n
		WHERE
			n.instance_id = 'COC'
			AND n.object = 'ENTITYGROUP'))
ORDER BY
	n.instance_id;
```

# 烘烤

遍历循环每个真空包，先查询真空包状态是否可以烘烤，再修改状态

```js
TransactionLog transactionLog = baseService.startTransactionLog(userName, PackedLot.TRANS_BAKE);  
for (Iterator<PackedLot> iterator = bakeList.iterator(); iterator.hasNext(); ) {  
    PackedLot packedLot = iterator.next();  
    PackedLot tbox = packedLotRepository.getPackedLotByBoxId(packedLot.getBoxId());  
    if (!StringUtils.equals(PackedLot.STATUS_PQC1, tbox.getPackedStatus())  
            && !StringUtils.equals(PackedLot.STATUS_BAKED, tbox.getPackedStatus())  
            && !StringUtils.equals(PackedLot.TRANS_PQC, tbox.getPackedStatus())) {  
        throw new MyCimException("此状态真空包不可被烘);  
    }  
    tbox.setFinalOperationTime(new Date());  
    tbox.setTransComment(packedLot.getTransComment());  
    tbox.setPackedStatus(PackedLot.STATUS_BAKED);  
    packedLotRepository.update(tbox);  
    savePackedCheckHistory(transactionLog.getTransRrn(), PackedLot.STATUS_BAKED, tbox);  
}  
baseService.markTransactionLog(transactionLog);
```

## 查询

```js
return getSqlBuilder().select(columnNames)  
        .from(tableName)  
        .where(" TYPE ").eq(PackedLot.TYPE_VBOX)  
        .and(" PRODUCT_CATEGORY ").eq(PackedLot.COM)  
        .and(" packed_status in ('PQC1', 'PQC') ")  
        .queryList(PackedLot.class);
```
