@Component
public class InventoryCountRepositoryImpl extends AbstractRepositoryImpl<InventoryCount, String> implements InventoryCountRepository {
    @Override
    public InventoryCount getInventoryCount(String lotId, String actionType) throws MyCimException {
        InventoryCount inventoryCount = null;
        try {
            if (StringUtils.isNotEmpty(lotId)) {

                String mmLotSql = " SELECT mpl.BOX_ID as SCANNING_NUMBER, mpl.PRODUCT_ID, mpl.QUANTITY, mpl.LEVEL_TWO_CODE, mpl.GRADE, mpl.BONDED_PROPERTY," +
                        " mpl.PACKED_STATUS as STATUS, mpl.WAFER_QTY as QTY1 FROM MM_PACKED_LOT mpl WHERE BOX_ID = '" + lotId + "' AND mpl.PRODUCT_CATEGORY = '" + actionType + "'";
                inventoryCount = getSqlBuilder().useSql(mmLotSql).queryOne(InventoryCount.class, false);
                
                if (inventoryCount == null){

                    String lotSql = " SELECT l.LOT_ID as SCANNING_NUMBER, no2.INSTANCE_ID AS PRODUCT_ID, l.QTY1, " +
                            " no3.INSTANCE_ID AS PROCESS_ID, l.LOT_STATUS as STATUS, " +
                            " REGEXP_SUBSTR(l.process_step_id_version,'[^,]+',instr(l.process_step_id_version,'|',1,2) + 1,1) as STEP, " +
                            " le.BONDED_PROPERTY, le.LEVEL_TWO_CODE FROM LOT l, LOT_EXT le, NAMED_OBJECT no2, NAMED_OBJECT no3 " +
                            " WHERE l.LOT_ID = '" + lotId +
                            "' AND l.LOT_RRN = le.LOT_RRN AND no2.INSTANCE_RRN = l.PRODUCT_RRN AND no3.INSTANCE_RRN = l.PROCESS_RRN AND l.LINE_TYPE = '" + actionType + "'";
                    inventoryCount = getSqlBuilder().useSql(lotSql).queryOne(InventoryCount.class, false);
                    
                    if (inventoryCount != null){
                        String lotBinIdSql = " SELECT BIN_ID FROM WIP_LOT_DIE_BIN wldb WHERE LOT_RRN = (SELECT LOT_RRN  FROM LOT WHERE LOT_ID = '" + lotId + "')";
                        List<String> binIds = getSqlBuilder().useSql(lotBinIdSql).queryList(String.class);
                        for (String binId : binIds) {
                            if (StringUtils.isNotEmpty(inventoryCount.getGrade())) {
                                inventoryCount.setGrade(inventoryCount.getGrade() + "," + binId);
                            } else {
                                inventoryCount.setGrade(binId);
                            }
                        }
                    }
                }
            }
            return inventoryCount;
        } catch (Exception e) {
            throw ExceptionHandler.handlerException(e);
        }
    }

}

Ext.doComInventoryCountSearch = function () {

    var sourceModel = Ext.getCmp('comInventoryCountGrid').getSelectionModel();
    var lot = Ext.getCmp('ScanningLot').getValue();
    if (Ext.isEmpty(lot)) {
        MyCim.notify.alert(i18n_msg_scan_id_is_not_empty);
        Ext.getCmp('ScanningLot').focus(false, 100);
        return;
    }
    if (lotIsAdded(lot)) {
        comInventoryCountStore.each(function (record, index) {
            if (record.data.scanningNumber == lot) {
                sourceModel.select(index, true, false);
            }
        });
        Ext.getCmp('ScanningLot').setValue('');
        Ext.getCmp('ScanningLot').focus(false, 100);
        return;
    }

    Ext.Ajax.request({
        url: '/mycim2/inventoryCount.do?action=search',
        params: {
            scanningLot: lot,
            actionType: "COM"
        },
        success: function (response) {
            var responseJson = Ext.JSON.decode(response.responseText);
            if (responseJson.success != null && responseJson.success) {
                var data = responseJson.msg.data;
                comInventoryCountStore.add(data);
                comInventoryCountStore.each(function (record, index) {
                    sourceModel.select(index, true, false);
                });
                Ext.getCmp('ScanningLot').setValue('');
                Ext.getCmp('ScanningLot').focus(false, 100);
            } else {
                MyCim.notify.alert(responseJson.msg);
            }
        }
    })

};

function lotIsAdded(lot) {
    var flag = false;
    comInventoryCountStore.each(function (record) {
        var lotId = record.data.scanningNumber;
        if (lotId == lot) {
            flag = true;
            return flag;
        }
    });
    return flag;
}

Ext.doComInventoryCountInsert = function () {
    var records = Ext.getCmp('comInventoryCountGrid').getSelectionModel().getSelection();
    if (records.length == 0) {
        MyCim.notify.alert(i18n_fld_Please_Select_The_Data);
        return;
    }
    var scannedLotList = new Array();
    Ext.Array.each(records, function (item) {
        scannedLotList.push(item.data);
    });

    Ext.Ajax.request({
        url: '/mycim2/inventoryCount.do?action=save',
        params: {
            scannedLotList: Ext.encode(scannedLotList),
            actionType: "COM"
        },
        success: function (response) {
            var responseJson = Ext.JSON.decode(response.responseText);
            if (responseJson.success != null && responseJson.success) {
                MyCim.notify.alert(i18n_msg_SuccessfulOperation, function () {
                    Ext.getCmp('ScanningLot').focus(false, 100);
                    comInventoryCountStore.removeAll();
                });
            } else {
                MyCim.notify.alert(responseJson.msg);
            }
        }
    });
}

Ext.doComInventoryCountClear = function () {
    Ext.getCmp('ScanningLot').setValue('');
    comInventoryCountStore.removeAll();
}

function initTabPanel() {

function tabPanelEmpower(mainTabPanel, targetTabPanel, markId) {
    Ext.Ajax.request({
        url: '/mycim2/ajaxflush.do?type=checkButtonPermission',
        params: {"markId": markId},//系统中定义的按钮
        method: 'POST',
        success: function (response, options) {
            var responseJson = Ext.JSON.decode(response.responseText);
            if (responseJson.success) {
            } else {
                var mainTabitems = mainTabPanel.items.items;
                Ext.Array.each(mainTabitems, function (item) {
                    if (item.items.items[0].id == targetTabPanel.id) {
                        mainTabPanel.remove(item);
                    }
                });
            }
        },
        failure: function (response, options) {
            showAlertByExt(i18n.msgs.MSG_FAILED, i18n_msg_RequestTimeoutOrNetworkFailureErrorCode + response.status, 1);
