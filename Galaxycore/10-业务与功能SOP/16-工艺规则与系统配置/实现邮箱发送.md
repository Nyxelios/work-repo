Ext.doAdd = function (){
	var receiver =  Ext.getCmp('receiver').getValue();
	if(Ext.isEmpty(receiver)){
		MyCim.notify.alert(i18n_msg_receiver_empty);
		return;
	}

	Ext.Ajax.request({
		url: '/mycim2/mailSetup.do?action=add',
		params: {
			receiver: receiver
		},
		success : function(response) {
			var responseJson = Ext.JSON.decode(response.responseText);
			if (responseJson.success != null && responseJson.success) {
				var data = responseJson.msg.data;
				store1.add(data);
			} else {
				MyCim.notify.alert(responseJson.msg);
			}
		}
	});
}

Ext.doClear = function (){
	Ext.getCmp('receiver').setValue('');
	store1.removeAll();
}

Ext.doSearch = function (){
	// var values = Ext.getCmp('mailSetupForm').getValues();
	// store1.proxy.extraParams = values;
	store1.load({
		doSuccessCallback : function(responseJson, operation, success) {
			if (responseJson.success) {
				Ext.getCmp('receiver').setValue('');
			}else {
				MyCim.notify.alert(responseJson.msg);
			}
		}
	});
};

Ext.doClearFrom = function (){
	Ext.getCmp('receiver').setRawValue('');
}

Ext.deleteProgram = function () {
	Ext.Msg.confirm(i18n_msg_OperationConfirmation, i18n_msg_ConfirmDeletion, function (btn) {
		if (btn == 'yes') {
			var records = Ext.getCmp('mailSetupGrid').getSelectionModel().getSelection();
			if (records.length == 0) {
				MyCim.notify.alert(i18n_fld_Please_Select_The_Data);
				return;
			}
			Ext.Array.each(records, function (item) {
				var objectRrn = item.data.objectRrn;
				Ext.Ajax.request({
					url: '/mycim2/mailSetup.do?action=delete',
					params: {
						objectRrn: objectRrn,
					},
					success: function (response) {
						store1.remove(item);
					}
				});
			});
		}
	}, this);
};

@Slf4j
@Transactional
@Service
public class MailSetupServiceImpl extends AbstractService implements MailSetupService {

    @Autowired
    private MailSetupRepository mailSetupRepository;
    private Logger LOGGER = LoggerFactory.getLogger(MailSetupServiceImpl.class);

    @Override
    public void addMailSetup(MailSetup mailSetup) throws MyCimException {
        try {
            mailSetup.setObjectRrn(getObjectRRN());
            mailSetupRepository.merge(mailSetup);
        } catch (Exception e) {
            throw ExceptionHandler.handlerException(e, log);
        }
    }

    @Override
    public List<MailSetup> getMails() throws MyCimException {
        String sql = "SELECT * FROM MAILBOX";
        return mailSetupRepository.useSqlQuery(sql, MailSetup.class);
    }

    public void sendEmail(GcModelManagement modelManagement, String old) throws MyCimException, IOException, MessagingException {

        Session session = createSession();

        //2.创建邮件对象
        MimeMessage message=new MimeMessage(session);
        MimeMessageHelper messageHelper = new MimeMessageHelper(message, true, "utf-8");
        //设置邮件主题
        message.setSubject("标准测试工时字段变更通知");
        //设置发件
        message.setFrom(new InternetAddress("cim_test@gcoreinc.com"));
        //邮件内容
        messageHelper.setText(getContent(modelManagement, old), true);
        //设置收件
        List<String> arrList = mailSetupRepository.useSqlQuery("SELECT RECEIVER FROM MAILBOX", String.class);
        String[] arr = arrList.toArray(new String[0]);
        messageHelper.setTo(arr);
        //3.发送邮
        Transport.send(message);
    }

    @Override
    public void deleteMailSetup(Long id) throws MyCimException {
        try {
            mailSetupRepository.deleteOneById(id);
        } catch (MyCimException e) {
            throw new MyCimException("邮箱ID不存);
        }
    }

    private String getContent(GcModelManagement modelManagement, String old) throws MyCimException {
        StringBuilder sb = new StringBuilder();
        try {
            sb.append("<html><head></head><table border=\\"0\\" cellspacing=\\"0\\" cellpadding=\\"5\\">");
            sb.append("<tr><td>入库型号/td><td>").append(escapeHtml(modelManagement.getInStorageModelId())).append("</td></tr>");
            sb.append("<tr><td>原型号：</td><td>").append(escapeHtml(modelManagement.getOriginalModelId())).append("</td></tr>");
            sb.append("<tr><td>测试型号/td><td>").append(escapeHtml(modelManagement.getTestModelId())).append("</td></tr>");
            sb.append("<tr><td>工艺流程号：</td><td>").append(escapeHtml(modelManagement.getRouteId())).append("</td></tr>");
            sb.append("<tr><td>Step ID/td><td>").append(escapeHtml(modelManagement.getStepId())).append("</td></tr>");
            sb.append("<tr><td>程序名：</td><td>").append(escapeHtml(modelManagement.getProgramId())).append("</td></tr>");
            sb.append("<tr><td>WLA测试位：</td><td>").append(escapeHtml(modelManagement.getWlaTestBit())).append("</td></tr>");
            sb.append("<tr><td>程序位：</td><td>").append(escapeHtml(modelManagement.getProgramBit())).append("</td></tr>");
            sb.append("<tr><td>标准测试时长（变更前）：</td><td>").append(escapeHtml(old)).append("</td></tr>");
            sb.append("<tr><td>标准测试时长（变更后）：</td><td>").append(escapeHtml(String.valueOf(modelManagement.getTestTime()))).append("</td></tr>");
            sb.append("</table></body></html>");
            return sb.toString();
        } catch (MyCimException e) {
            LOGGER.error("Error generating content: ", e); // 记录错误日志
            throw new MyCimException(e.getMessage());
        } catch (Exception e) { // 增加对未预期异常的处
            LOGGER.error("Unexpected error occurred: ", e);
            throw new MyCimException("Unexpected error occurred while generating content.");
        }
    }

    private String escapeHtml(String input) {
        if (input == null) {
            return null;
        }
        return StringEscapeUtils.escapeHtml(input);
    }

    private Session createSession() {

        //	创建一个配置文件，并保
        Properties props = new Properties();

        //	SMTP服务器连接信

package com.mycim.webapp.actions.prp;

import com.mycim.AppContext;
import com.mycim.core.json.JSONUtils;
import com.mycim.core.json.ReponseJSONBuilder;
import com.mycim.tool.model.MailSetup;
import com.mycim.tool.service.MailSetupService;
import com.mycim.utils.ThreadLocalContext;
import com.mycim.webapp.Constants;
import com.mycim.webapp.WebUtil;
import com.mycim.webapp.actions.PrpSetupAction;
import org.apache.struts.action.ActionForm;
import org.apache.struts.action.ActionForward;
import org.apache.struts.action.ActionMapping;

import javax.servlet.ServletException;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import java.util.HashMap;
import java.util.Map;

public class MailSetupAction extends PrpSetupAction {

    private MailSetupService mailSetupService = AppContext.getBean(MailSetupService.class);

    public static final String ACTION_COMBO_QUERY = "query";
    public static final String ACTION_COMBO_ADD = "add";
    public static final String ACTION_COMBO_DELETE = "delete";

    public ActionForward perform(ActionMapping mapping, ActionForm form, HttpServletRequest request,
                                 HttpServletResponse response) throws ServletException {
        String userName = ThreadLocalContext.getUsername();
        if (userName == null) {
            return mapping.findForward("login");
        }
        String action = request.getParameter("action");

        if (ACTION_COMBO_QUERY.equals(action)) {
            return query(response);
        }else if (ACTION_COMBO_ADD.equals(action)) {
            return add(request, response);
        }else if (ACTION_COMBO_DELETE.equals(action)) {
            return delete(request, response);
        }
        return mapping.findForward(Constants.MODIFY_KEY);
    }

    private ActionForward query(HttpServletResponse response) {
        Map<String, Object> map = new HashMap<>();
        try {
            map.put("data", mailSetupService.getMails());
            WebUtil.writeJson(response, ReponseJSONBuilder.buildSuccessMsg(JSONUtils.toJSONString(map)));
        } catch (Exception e) {
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));
        }
        return WebUtil.NULLActionForward;
    }

    private ActionForward delete(HttpServletRequest request, HttpServletResponse response) {
        Map<String, Object> map = new HashMap<>();
        try {
            String objectRrn = request.getParameter("objectRrn");
            mailSetupService.deleteMailSetup(Long.valueOf(objectRrn));
            WebUtil.writeJson(response, ReponseJSONBuilder.buildSuccessMsg(JSONUtils.toJSONString(map)));
        } catch (Exception e) {
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));
        }
        return WebUtil.NULLActionForward;
    }

    private ActionForward add(HttpServletRequest request, HttpServletResponse response) {
        Map<String, Object> map = new HashMap<>();
        try {
            String receiver = request.getParameter("receiver");
            MailSetup mailSetup = new MailSetup();
            mailSetup.setReceiver(receiver);
            mailSetupService.addMailSetup(mailSetup);
            map.put("data", mailSetup);
            WebUtil.writeJson(response, ReponseJSONBuilder.buildSuccessMsg(JSONUtils.toJSONString(map)));
        } catch (Exception e) {
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));
        }
        return WebUtil.NULLActionForward;
    }
    
}

/**
 * @ClassName A
 * @Description: 类描
 * @Author: natsume_wang
 * @CreateDate: 2024/7/10 10:16
 * @Version: 1.0
 */
public class JavaMailUntil {

    public static Session createSession() {

        //	账号信息
        String username = "cim_test@gcoreinc.com";//	邮箱发送账
        String password = "GC@123456";//	邮箱授权

        //	创建一个配置文件，并保
        Properties props = new Properties();

        //	SMTP服务器连接信
        //  126——smtp.126.com
        //  163——smtp.163.com
        //  qq——smtp.qq.com"
        props.put("mail.smtp.host", "GCEXSRV01.gcoreinc.com");//	SMTP主机

        //  126—5
        //  163—65 163也可以用5
        props.put("mail.smtp.port", "25");//	主机端口
        props.put("mail.smtp.auth", "false");//	是否需要用户认
        props.put("mail.smtp.starttls.enale", "true");//	启用TlS加密
        props.setProperty("mail.smtp.timeout", "1000");

        //	创建session会话
        //	参数1:smtp服务器连接参
        //	参数2:账号和密码的授权认证对象
        Session session = Session.getInstance(props, new Authenticator() {
            @Override
            protected PasswordAuthentication getPasswordAuthentication() {
                // TODO Auto-generated method stub
                return new PasswordAuthentication(username, password);
            }
        });
        //  控制台打印调试信
        session.setDebug(true);
        return session;
    }
}
