public class ComWaferRegexConfigAction extends PrpSetupAction {  
    private static final String ACTION_CREATE = "create";  
    private static final String ACTION_UPDATE = "update";  
    private static final String ACTION_READ = "read";  
    private static final String ACTION_DELETE = "delete";  
    String ACTION_PRODUCTIDCOMBOBOX = "productIdComboBox";  
    private final WaferBindSyncService waferBindSyncService = AppContext.getBean(WaferBindSyncService.class);  
  
    public ActionForward perform(ActionMapping mapping, ActionForm form, HttpServletRequest request,  
                                 HttpServletResponse response) throws ServletException {  
        String userName = ThreadLocalContext.getUsername();  
        Long facilityRrn = ThreadLocalContext.getFacilityRrn();  
        if (userName == null) {  
            return mapping.findForward(Constants.LOGIN_KEY);  
        }  
        String action = request.getParameter("action");  
  
        if(StringUtils.equals(ACTION_CREATE, action)){  
            return create(request, response);  
        }else if(ACTION_UPDATE.equals(action)){  
            return update(request, response);  
        }else if(ACTION_READ.equals(action)){  
            return read(request, response);  
        }else if(ACTION_DELETE.equals(action)){  
            return delete(request, response);  
        } else if (StringUtils.isEqual(action, ACTION_PRODUCTIDCOMBOBOX)) {  
            return queryProductIdComboBox(request, response, Lists.newArrayList(WorkOrder.COM));  
        } else if (StringUtils.equals("importFile", action)) {  
            return importFile(form, response);  
        }  
        return mapping.findForward(Constants.MODIFY_KEY);  
    }  
  
    private ActionForward create(HttpServletRequest request, HttpServletResponse response) {  
        try {  
            ComWaferRegexConfig config = (ComWaferRegexConfig) HttpUtils.getRequestConvertBean(request, ComWaferRegexConfig.class);  
            comServices.createComWaferRegexConfig(config);  
            writeDefaultResponseJson(config);  
        } catch (Exception e) {  
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));  
        }  
        return WebUtil.NULLActionForward;  
    }  
  
    private ActionForward update(HttpServletRequest request, HttpServletResponse response) {  
        try {  
            ComWaferRegexConfig config = (ComWaferRegexConfig) HttpUtils.getRequestConvertBean(request, ComWaferRegexConfig.class);  
            comServices.updateComWaferRegexConfig(config);  
            writeDefaultResponseJson();  
        } catch (Exception e) {  
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));  
        }  
        return WebUtil.NULLActionForward;  
    }  
  
    private ActionForward read(HttpServletRequest request, HttpServletResponse response) {  
        try {  
            ComWaferRegexConfig config = (ComWaferRegexConfig) HttpUtils.getRequestConvertBean(request, ComWaferRegexConfig.class);  
            ResultPage<ComWaferRegexConfig> resultPage = new ResultPage<>();  
            resultPage = comServices.readComWaferRegexConfig(config, getPageNoForExt(request), getPageSizeForExt(request));  
            for (ComWaferRegexConfig datum : resultPage.getContent()) {  
                List<WmsMmsMaterialLotUnit> wmsMmsMaterialLotUnits = waferBindSyncService.searchWms(null, null, config.getBoxId());  
                BigDecimal qty = BigDecimal.valueOf(0);  
                for (WmsMmsMaterialLotUnit wmsMmsMaterialLotUnit : wmsMmsMaterialLotUnits) {  
                    qty = qty.add(wmsMmsMaterialLotUnit.getCurrentQty());  
                    datum.setDevice(wmsMmsMaterialLotUnit.getMaterialName());  
                }  
                datum.setWaferQty(qty);  
            }  
            writePageJson(resultPage);  
        } catch (Exception e) {  
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));  
        }  
        return WebUtil.NULLActionForward;  
    }  
  
    private ActionForward delete(HttpServletRequest request, HttpServletResponse response) {  
        try {  
            String objectRrnStr = request.getParameter("deleteList");  
            objectRrnStr = objectRrnStr.substring(1,objectRrnStr.length()-1);  
            String[] objectRrnList = objectRrnStr.split(",");  
            for (String str : objectRrnList) {  
                Long objectRrn = Long.valueOf(str);  
                comServices.deleteComWaferRegexConfig(objectRrn);  
            }  
            writeDefaultResponseJson();  
            return WebUtil.NULLActionForward;  
        } catch (Exception e) {  
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));  
        }  
        return WebUtil.NULLActionForward;  
    }  
  
    private ActionForward importFile(ActionForm form, HttpServletResponse response) {  
        try {  
            ImportFileForm theform = (ImportFileForm) form;  
            String fileName = theform.getFormFile().getFileName();  
            if (StringUtils.isEmpty(fileName)) {  
                throw new MyCimException(PrpExceptions.I_PLEASE_SELECT_EXCEL_FILE);  
            }  
            String nameSuffix = StringUtils.substringAfterLast(fileName, ".");  
            if (!(StringUtils.equalsIgnoreCase(CsvUtils.EXCEL_SUFFIX_CSV, nameSuffix))) {  
                throw new MyCimParameterException(PrpExceptions.I_IMPORT_FILE_SUFFIX_NAME_FORMAT_ERROR, fileName);  
            }  
            Map<String, String> headersMapped = Maps.newHashMap();  
            headersMapped.put("BOX_ID", "boxId");  
            headersMapped.put("PRODUCT_ID", "tProductId");  
            List<ComWaferRegexConfig> importDataList = (List<ComWaferRegexConfig>) CsvUtils  
                    .readCsvFile(headersMapped, ComWaferRegexConfig.class, theform.getFormFile().getInputStream(), ",");  
            if (CollectionUtils.isEmpty(importDataList)) {  
                throw new MyCimParameterException(PrpExceptions.I_IMPORT_FILE_HAVE_NO_DATA, fileName);  
            }  
            for (ComWaferRegexConfig comWaferRegexConfig : importDataList) {  
                comServices.createComWaferRegexConfig(comWaferRegexConfig);  
            }  
            writeDefaultResponseJson();  
            return WebUtil.NULLActionForward;  
        } catch (Exception e) {  
            WebUtil.writeJson(response, ReponseJSONBuilder.buildErrorMsg(e));  
        }  
        return WebUtil.NULLActionForward;  
    }  
  
}

            config.setType(ComWaferRegexConfig.TYPE_BOX);

        }

        throw ExceptionHandler.handlerException(e, log);  
    }  
}

@Component  
public class ComWaferRegexConfigRepositoryImpl extends AbstractRepositoryImpl<ComWaferRegexConfig, Long> implements ComWaferRegexConfigRepository {  
  
    /**  
     * @param productId  
     * @param lotId  
     * @return  
     */  
    @Override  
    public boolean judgeBindByBox(String productId, String lotId) {  
        try {  
            SqlBuilder sqlBuilder = getSqlBuilder().select(columnNames).from(tableName).where("1").eq(1);  
            if (StringUtils.isNotEmpty(lotId)) {  
                sqlBuilder.and(" BOX_ID ").eq(lotId);  
            }  
            sqlBuilder.and(" TYPE ").eq(ComWaferRegexConfig.TYPE_BOX);  
            List<ComWaferRegexConfig> comWaferRegexConfigs = sqlBuilder.queryList(ComWaferRegexConfig.class);  
            if (comWaferRegexConfigs.isEmpty()) {  
                return true;  
            } else {  
                for (ComWaferRegexConfig comWaferRegexConfig : comWaferRegexConfigs) {  
                    if (StringUtils.equals(comWaferRegexConfig.gettProductId(), productId)) {  
                        return true;  
                    }  
                }  
                return false;  
            }  
        } catch (Exception e) {  
            throw new MyCimException(e);  
        }  
    }  
  
    /**  
     * @return  
     */  
    @Override  
    public ResultPage<ComWaferRegexConfig> readComWaferRegexConfig(ComWaferRegexConfig config, int pageNoForExt, int pageSizeForExt) {  
        try {  
            SqlBuilder sqlBuilder = getSqlBuilder().select(columnNames).from(tableName).where("1").eq(1);  
            if (StringUtils.isNotEmpty(config.gettProductId())) {  
                sqlBuilder.and(" T_PRODUCT_ID ").eq(config.gettProductId());  
            }  
            if (StringUtils.isNotEmpty(config.getBoxId())) {  
                sqlBuilder.and(" BOX_ID ").eq(config.getBoxId());  
            }  
            sqlBuilder.and(" TYPE ").eq(ComWaferRegexConfig.TYPE_BOX);  
            return sqlBuilder.queryPage(pageNoForExt, pageSizeForExt, ComWaferRegexConfig.class);  
        } catch (Exception e) {  
            throw new MyCimException(e);  
        }  
    }  
}
