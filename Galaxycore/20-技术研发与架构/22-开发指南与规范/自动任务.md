# applicationContext-quartz.xml

```jsx
<!-- UPDATE WAFER BIND STATUS -->  
<bean name="updateWaferBindSyncJobDetail"  class="org.springframework.scheduling.quartz.JobDetailFactoryBean">  
    <property name="jobClass" value="com.mycim.tool.service.quartz.jobs.WaferBindSyncJob" />  
    <property name="jobDataMap">  
        <map>            <entry key="service" value="timelimit"/>  
        </map>    </property>    <property name="durability" value="true" />  
</bean>  
<bean id="updateWaferBindSyncJobCronTrigger"  class="org.springframework.scheduling.quartz.CronTriggerFactoryBean">  
    <property name="jobDetail" ref="updateWaferBindSyncJobDetail" />  
    <property name="cronExpression" value="0 0/30 7-23 * * ?" />  
</bean>
```
