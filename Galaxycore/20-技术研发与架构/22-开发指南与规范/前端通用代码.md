function initTabPanel() {  
    var ComWaferRegexConfigTabPanel = Ext.getCmp("ComWaferRegexConfigTabPanel");  
    var ComWaferRegexConfigTab = Ext.getCmp("ComWaferRegexConfigTab");  
    tabPanelEmpower(ComWaferRegexConfigTabPanel, ComWaferRegexConfigTab);  
}  
  
function tabPanelEmpower(mainTabPanel, targetTabPanel, markId) {  
    Ext.Ajax.request({  
        url: '/mycim2/ajaxflush.do?type=checkButtonPermission',  
        params: {"markId": markId},//系统中定义的按钮，    
method: 'POST',  
        success: function (response, options) {  
            var responseJson = Ext.JSON.decode(response.responseText);  
            if (responseJson.success) {  
            } else {  
                var mainTabitems = mainTabPanel.items.items;  
                Ext.Array.each(mainTabitems, function (item) {  
                    if (item.items.items[0].id == targetTabPanel.id) {  
                        mainTabPanel.remove(item);  
                    }  
                });  
            }  
        },  
        failure: function (response, options) {  
            showAlertByExt(i18n.msgs.MSG_FAILED, i18n_msg_RequestTimeoutOrNetworkFailureErrorCode + response.status, 1);  
        }  
    });  
}

Ext.doRead = function () {

Ext.doDelete = function () {  
    var records = Ext.getCmp("ComWaferRegexConfigGrid").getSelectionModel().getSelection();

        deleteList.push(item.data.objectRrn);

            return;        }

        if (records.length == 0) {

        CpMpProductConfigStore.proxy.extraParams = Ext.getCmp('CpMpProductConfigForm').getValues();  
        CpMpProductConfigStore.load({  
            url: '/mycim2/CpMpProductConfig.do?action=update',

                    CpMpProductConfigStore.load();

Ext.doDelete = function () {  
    Ext.Msg.confirm(i18n_msg_OperationConfirmation, i18n_msg_ConfirmDeletion, function (btn) {  
        if (btn == 'yes') {  
            var records = Ext.getCmp('CpMpProductConfigGrid').getSelectionModel().getSelection();  
            if (records.length == 0) {  
                MyCim.notify.alert(i18n_fld_Please_Select_The_Data);  
                return;  
            }  
            Ext.Array.each(records, function (item) {  
                var objectRrn = item.data.objectRrn;  
                Ext.Ajax.request({  
                    url: '/mycim2/CpMpProductConfig.do?action=delete',  
                    params: {  
                        objectRrn: objectRrn  
                    },  
                    success: function (response) {  
                        var responseJson = Ext.JSON.decode(response.responseText);  
                        if (responseJson.success != null && responseJson.success) {  
                            CpMpProductConfigStore.remove(item);  
                        } else {  
                            MyCim.notify.alert(responseJson.msg);  
                        }  
                    }  
                });  
            });  
        }  
    }, this);  
};

function tabPanelEmpower(mainTabPanel, targetTabPanel, markId) {  
    Ext.Ajax.request({  
        url: '/mycim2/ajaxflush.do?type=checkButtonPermission',  
        params: {"markId": markId},//系统中定义的按钮，  
        method: 'POST',  
        success: function (response, options) {  
            var responseJson = Ext.JSON.decode(response.responseText);  
            if (responseJson.success) {  
            } else {  
                var mainTabitems = mainTabPanel.items.items;  
                Ext.Array.each(mainTabitems, function (item) {  
                    if (item.items.items[0].id == targetTabPanel.id) {  
                        mainTabPanel.remove(item);  
                    }  
                });  
            }  
        },  
        failure: function (response, options) {  
            showAlertByExt(i18n.msgs.MSG_FAILED, i18n_msg_RequestTimeoutOrNetworkFailureErrorCode + response.status, 1);  
        }  
    });  
}
