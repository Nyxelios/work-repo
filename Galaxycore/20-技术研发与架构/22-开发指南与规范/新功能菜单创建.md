    Ext.doAdd = function (){
    	var materialGrade =  Ext.getCmp('materialGrade').getValue();
    	var gradeDefinition =  Ext.getCmp('gradeDefinition').getValue();
    	var testProgram =  Ext.getCmp('testProgram').getValue();
    	var businessGrade =  Ext.getCmp('materialGrade').getValue();
    	//var activeFlag =  Ext.getCmp('activeFlag').getValue();
    	if(Ext.isEmpty(materialGrade) || Ext.isEmpty(gradeDefinition) 
    			|| Ext.isEmpty(testProgram) || Ext.isEmpty(businessGrade)){
    		MyCim.notify.alert(i18n_msg_RequiredFiledValuesCannotBeEmpty);
    		return;
    	}
    	Ext.Ajax.request({
    		url: '/mycim2/materialBinGlance.do?action=addInfo',
    		params: {
    			materialGrade: materialGrade,
    			gradeDefinition: gradeDefinition,
    			testProgram: testProgram,
    			businessGrade: businessGrade,
    			//activeFlag: activeFlag
    		},

    			var responseJson = Ext.JSON.decode(response.responseText);

    				var data = responseJson.msg.data;
    				store1.add(data);
    			} else {
    				MyCim.notify.alert(responseJson.msg);
    			}
    		}
    	});

    	var records = Ext.getCmp("MaterialBinGlanceGrid").getSelectionModel().getSelection();
    	if(records.length == 0){
    		Ext.Msg.alert(i18n_msg_prompt, i18n_msg_PLEASE_CHECK_AT_LEAST_ONE_PIECE_OF_DATA);
    	    return;
    	}
    	var stopList = new Array();
    	Ext.Array.each(records, function(item) {
    		stopList.push(item.data.objectRrn);
        });
        Ext.Msg.confirm(i18n_msg_OperationConfirmation,i18n_msg_ConfirmStop, function(btn) {
    		if (btn == 'yes') {
    			Ext.Ajax.request({
    				url: '/mycim2/materialBinGlance.do?action=stop',
    				params: {
    					stopList:stopList
    				},
    				success: function(response) {
    					Ext.doSearch();
    				}
    			  });
    			return;
    		} 
    	}, this);
