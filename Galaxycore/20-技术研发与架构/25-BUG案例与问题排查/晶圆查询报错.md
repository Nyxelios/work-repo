@Override
	public List<WaferReceive> getByFacilityRrnAndCstId(Long facilityRrn, String cstId, String status) throws MyCimException {
		try {
			SqlBuilder sqlBuilder = getSqlBuilder().select(columnNames).from(tableName)
					.where("FACILITY_RRN").eq(facilityRrn);
			if(StringUtils.isNotEmpty(cstId)){
				sqlBuilder.and("CST_ID").eq(cstId);
			}
			if(StringUtils.isNotEmpty(status)){
				sqlBuilder.and("STATUS").eq(status);
			}
			sqlBuilder.and("WAFER_SOURCE").eq("16");
			return sqlBuilder.queryList(WaferReceive.class);
		} catch (Exception e) {
			throw ExceptionHandler.handlerException(e, log);
		}
	}
